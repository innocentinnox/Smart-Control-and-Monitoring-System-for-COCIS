% \screenshot[width]{file}{caption}{what to capture}
%   Shows figures/<file>.png if it exists; otherwise a red box saying what to capture,
%   so a missing screenshot is obvious in the PDF but never breaks the build.
\newcommand{\screenshot}[4][\linewidth]{%
  \begin{figure}[H]
  \centering
  \IfFileExists{figures/#2.png}{\includegraphics[width=#1,height=0.75\textheight,keepaspectratio]{#2.png}}{%
    \fcolorbox{red}{red!4}{\parbox{0.9\linewidth}{\centering\vspace{1.5em}
      \textcolor{red}{\textbf{Screenshot needed:} \texttt{docs/report/figures/#2.png}}\\[0.4em]
      \small\detokenize{#4}\vspace{1.5em}}}}
  \caption{#3}
  \label{fig:#2}
  \end{figure}}

% \consolefig{file}{caption}: real serial output captured from a Wokwi run
%   (figures/console/<file>.txt, made by the scenarios in firmware/test/wokwi/)
\newcommand{\consolefig}[2]{%
  \begin{figure}[H]
  \lstinputlisting[style=console]{figures/console/#1.txt}
  \caption{#2}
  \label{fig:con-#1}
  \end{figure}}
